//Para acceder a la rama main
git branch -M main

//Para subir la carpeta a la rama main
git push -u origin main

//Para descargar las ramas que se crearon en GitHub
git fetch origin

//Para verificar las ramas existentes
git branch -a

//Para cambiar de rama en el que se subirán tus cambios
git chekout *nombre de la rama*

//Para subir tus cambios después del 'git add .' y 'git commit -m "" '. Solo podras ejecutar el comando cuando hallas hecho checkout en una rama

//Para descargar cambios que hay en GitHub
